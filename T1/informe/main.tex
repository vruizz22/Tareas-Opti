\section*{Pregunta 1: Planificaci\'on de producci\'on de aleaciones}
\subsection*{Conjuntos}
\begin{itemize}[leftmargin=2.2cm]
    \item[$\mathcal{J} = \{1,\dots,J\}$:] conjunto de tipos de aleaciones de oro.
    \item[$\mathcal{M} = \{1,\dots,M\}$:] conjunto de metales base disponibles.
    \item[$\mathcal{T} = \{1,\dots,T\}$:] conjunto de per\'iodos del horizonte de planificaci\'on.
\end{itemize}

\subsection*{Par\'ametros}
\begin{itemize}[leftmargin=2.2cm]
    \item[$d_{jt}$:] demanda m\'axima (gramos) de la aleaci\'on $j\in\mathcal{J}$ en el per\'iodo $t\in\mathcal{T}$.
    \item[$p_{jt}$:] precio de venta (\$/gramo) de la aleaci\'on $j$ en el per\'iodo $t$.
    \item[$r_m$:] \'indice de calidad del metal $m\in\mathcal{M}$.
    \item[$q_j$:] calidad promedio ponderada m\'inima exigida a la aleaci\'on $j$.
    \item[$a_{mt}$:] gramos del metal $m$ que se reciben al inicio del per\'iodo $t$.
    \item[$\widehat{\gamma}_m$:] inventario inicial (gramos) del metal $m$ (al inicio del per\'iodo $1$).
    \item[$\gamma_j$:] inventario inicial (gramos) de la aleaci\'on $j$.
    \item[$\widehat{h}_{mt}$:] costo (\$/gramo) de mantener el metal $m$ en inventario desde $t$ hasta $t+1$.
    \item[$h_{jt}$:] costo (\$/gramo) de mantener la aleaci\'on $j$ en inventario desde $t$ hasta $t+1$.
    \item[$v_m$:] valor (\$/gramo) del metal $m$ que quede en inventario al final del horizonte.
\end{itemize}

\subsection*{Variables de decisi\'on}
Todas las variables son continuas y no negativas (gramos o d\'olares, seg\'un corresponda).
\begin{itemize}[leftmargin=2.4cm]
    \item[$x_{mjt}$:] gramos del metal $m$ utilizados para producir la aleaci\'on $j$ en el per\'iodo $t$.
    \item[$y_{jt}$:] gramos producidos de la aleaci\'on $j$ en el per\'iodo $t$.
    \item[$w_{jt}$:] gramos vendidos de la aleaci\'on $j$ en el per\'iodo $t$.
    \item[$I^{M}_{mt}$:] gramos del metal $m$ en inventario al \emph{final} del per\'iodo $t$.
    \item[$I^{A}_{jt}$:] gramos de la aleaci\'on $j$ en inventario al \emph{final} del per\'iodo $t$.
\end{itemize}

\subsection*{Funci\'on objetivo}
Maximizamos la utilidad total: ingresos por ventas menos costos de mantener inventario (de metal y aleaci\'on), m\'as el valor de salvataje del metal al final del horizonte. (Los par\'ametros $a_{mt}$ y los inventarios iniciales son datos y no se incluyen como costos, pues ya ocurrieron o son ingresos del sistema.)

\[
\max\;\;
\underbrace{\sum_{j\in\mathcal{J}}\sum_{t\in\mathcal{T}} p_{jt}\, w_{jt}}_{\text{ingresos por ventas}}
\;-\;
\underbrace{\sum_{m\in\mathcal{M}}\sum_{t=1}^{T-1}\widehat{h}_{mt}\,I^{M}_{mt}}_{\text{costo inv. de metales}}
\;-\;
\underbrace{\sum_{j\in\mathcal{J}}\sum_{t=1}^{T-1} h_{jt}\,I^{A}_{jt}}_{\text{costo inv. de aleaciones}}
\;+\;
\underbrace{\sum_{m\in\mathcal{M}} v_m\,I^{M}_{mT}}_{\text{valor de salvataje}}.
\]

\subsection*{Restricciones}

\noindent\textbf{(R1) Balance de inventario de metales.} Para cada metal $m\in\mathcal{M}$ y per\'iodo $t\in\mathcal{T}$:
\[
I^{M}_{mt} \;=\; I^{M}_{m,\,t-1} + a_{mt} - \sum_{j\in\mathcal{J}} x_{mjt},
\qquad\text{con } I^{M}_{m,0} := \widehat{\gamma}_m.
\]
Es decir, lo que queda en bodega al final de $t$ es el inventario previo m\'as lo recibido, menos lo consumido en producci\'on.

\medskip
\noindent\textbf{(R2) Balance de inventario de aleaciones.} Para cada $j\in\mathcal{J}$, $t\in\mathcal{T}$:
\[
I^{A}_{jt} \;=\; I^{A}_{j,\,t-1} + y_{jt} - w_{jt},
\qquad\text{con } I^{A}_{j,0} := \gamma_j.
\]

\medskip
\noindent\textbf{(R3) Composici\'on de cada aleaci\'on.} La suma de los gramos de metales usados para producir la aleaci\'on $j$ debe ser \emph{exactamente} $y_{jt}$:
\[
\sum_{m\in\mathcal{M}} x_{mjt} \;=\; y_{jt}, \qquad \forall\, j\in\mathcal{J},\; t\in\mathcal{T}.
\]

\medskip
\noindent\textbf{(R4) Calidad promedio ponderada.} La calidad media ponderada de la aleaci\'on $j$ producida en $t$ debe ser al menos $q_j$. Cuando $y_{jt}>0$ equivale a $\sum_m r_m x_{mjt}/y_{jt} \geq q_j$; al multiplicar por $y_{jt}$ y usar la restricci\'on (R3), la linealizaci\'on correcta es:
\[
\sum_{m\in\mathcal{M}} (r_m - q_j)\,x_{mjt} \;\geq\; 0, \qquad \forall\, j\in\mathcal{J},\; t\in\mathcal{T}.
\]
(Si $y_{jt}=0$ la desigualdad se cumple trivialmente porque todos los $x_{mjt}=0$ por (R3).)

\medskip
\noindent\textbf{(R5) Cota de demanda.} No se puede vender m\'as que la demanda m\'axima:
\[
w_{jt} \;\leq\; d_{jt}, \qquad \forall\, j\in\mathcal{J},\; t\in\mathcal{T}.
\]

\medskip
\noindent\textbf{(R6) No negatividad.}
\[
x_{mjt},\, y_{jt},\, w_{jt},\, I^{M}_{mt},\, I^{A}_{jt} \;\geq\; 0, \qquad \forall\, m,j,t.
\]

\medskip

\noindent\emph{Observaci\'on.} Las variables $I^{M}_{mt}$ e $I^{A}_{jt}$ no son estrictamente necesarias (podr\'ian despejarse recursivamente de los balances); se mantienen para mayor claridad y porque aparecen tanto en la FO como en la restricci\'on de salvataje ($v_m I^M_{mT}$). El modelo completo es lineal continuo, tal como pide el enunciado.

\section*{Pregunta 2: Planificaci\'on de sesiones de estudio}

\subsection*{Conjuntos}
\begin{itemize}[leftmargin=2.2cm]
    \item[$\mathcal{T}=\{1,\dots,T\}$:] bloques horarios del d\'ia, ordenados.
    \item[$\mathcal{B}\subseteq\mathcal{T}$:] bloques en los que \emph{no} se puede estudiar.
\end{itemize}

\subsection*{Par\'ametros}
\begin{itemize}[leftmargin=2.0cm]
    \item[$h_t$:] costo fijo de \emph{iniciar} una sesi\'on al comienzo del bloque $t$.
    \item[$c_t$:] costo variable por bloque de estar estudiando en el bloque $t$.
    \item[$H$:] n\'umero m\'inimo total de bloques de estudio en el d\'ia.
    \item[$U$:] duraci\'on m\'inima de cualquier sesi\'on (bloques).
    \item[$D$:] descanso m\'inimo entre el t\'ermino de una sesi\'on y el inicio de la siguiente (bloques).
    \item[$K$:] duraci\'on m\'inima requerida para \emph{al menos una} sesi\'on del d\'ia, con $K>U$.
\end{itemize}

\subsection*{Variables de decisi\'on}
Todas son binarias.
\begin{itemize}[leftmargin=2cm]
    \item[$x_t$:] $1$ si el estudiante est\'a estudiando en el bloque $t$, $0$ en caso contrario.
    \item[$s_t$:] $1$ si \emph{inicia} una sesi\'on al comienzo del bloque $t$.
    \item[$e_t$:] $1$ si \emph{termina} una sesi\'on durante el bloque $t$ (es decir, $t$ es el \'ultimo bloque de una sesi\'on).
    \item[$z_t$:] $1$ si la sesi\'on que inicia en el bloque $t$ dura al menos $K$ bloques (variable auxiliar para la sesi\'on ``larga'').
\end{itemize}

\subsection*{Funci\'on objetivo}
Se minimiza el costo total: costos fijos por iniciar sesiones $+$ costos variables por bloque estudiado.
\[
\min \;\; \sum_{t\in\mathcal{T}} h_t\, s_t \;+\; \sum_{t\in\mathcal{T}} c_t\, x_t.
\]

\subsection*{Restricciones}

\medskip
\noindent\textbf{(R1) Relaci\'on entre $x$, $s$ y $e$.}  Notemos que $s_t=1$ equivale a ``estoy estudiando en $t$ pero no en $t-1$'' y $e_t=1$ equivale a ``estoy estudiando en $t$ pero no en $t+1$''. Con la convenci\'on $x_0=x_{T+1}=0$, podemos caracterizar estas variables mediante las siguientes dos familias de desigualdades:
\begin{align}
s_t &\;\geq\; x_t - x_{t-1}, & \forall\, t\in\mathcal{T}, \quad x_0 := 0, \label{eq:si1}\\
s_t &\;\leq\; x_t,            & \forall\, t\in\mathcal{T}, \label{eq:si2}\\
s_t &\;\leq\; 1 - x_{t-1},    & \forall\, t\in\mathcal{T}, \quad x_0 := 0, \label{eq:si3}\\
e_t &\;\geq\; x_t - x_{t+1}, & \forall\, t\in\mathcal{T}, \quad x_{T+1} := 0, \label{eq:ei1}\\
e_t &\;\leq\; x_t,            & \forall\, t\in\mathcal{T}, \label{eq:ei2}\\
e_t &\;\leq\; 1 - x_{t+1},    & \forall\, t\in\mathcal{T}, \quad x_{T+1} := 0. \label{eq:ei3}
\end{align}
En conjunto, esas desigualdades fuerzan $s_t = \max\{0, x_t - x_{t-1}\}$ y $e_t = \max\{0, x_t - x_{t+1}\}$.\footnote{Para $t=1$ se toma $x_0=0$, luego $s_1 = x_1$: si se empieza estudiando el d\'ia, se considera un inicio. Para $t=T$ se toma $x_{T+1}=0$, luego $e_T=x_T$: si se est\'a estudiando en el \'ultimo bloque, la sesi\'on termina en $T$, tal como indica el enunciado.}

\medskip
\noindent\textbf{(R2) Bloques prohibidos.}
\[
x_t \;=\; 0, \qquad \forall\, t\in\mathcal{B}.
\]

\medskip
\noindent\textbf{(R3) Total m\'inimo de bloques de estudio.}
\[
\sum_{t\in\mathcal{T}} x_t \;\geq\; H.
\]

\medskip
\noindent\textbf{(R4) Duraci\'on m\'inima $U$ de cada sesi\'on.} Si se inicia una sesi\'on en $t$, entonces se estudia durante los $U$ bloques consecutivos $t, t+1, \dots, t+U-1$:
\[
x_{t+k} \;\geq\; s_t, \qquad \forall\, t\in\mathcal{T},\; k=0,1,\dots,U-1 \text{ tal que } t+k\leq T.
\]
\emph{Cuidado con los bordes:} si $t>T-U+1$, entonces $t$ no alcanza a ``caber'' una sesi\'on de largo $U$; en ese caso prohibimos iniciar all\'i:
\[
s_t \;=\; 0, \qquad \forall\, t > T-U+1.
\]

\medskip
\noindent\textbf{(R5) Descanso m\'inimo $D$ entre sesiones.} Si termina una sesi\'on en $t$ ($e_t=1$), no puede estudiarse durante los $D$ bloques siguientes:
\[
x_{t+k} \;\leq\; 1 - e_t, \qquad \forall\, t\in\mathcal{T},\; k=1,2,\dots,D \text{ tal que } t+k\leq T.
\]

\medskip
\noindent\textbf{(R6) Al menos una sesi\'on de duraci\'on $\geq K$.} Introducimos $z_t$ que ``marca'' si la sesi\'on que inicia en $t$ dura $\geq K$. Dos familias de restricciones:
\begin{align}
z_t &\;\leq\; s_t, \qquad \forall\, t\in\mathcal{T}, \label{eq:z1}\\
x_{t+k} &\;\geq\; z_t, \qquad \forall\, t\in\mathcal{T},\; k=0,1,\dots,K-1 \text{ tal que } t+k\leq T. \label{eq:z2}
\end{align}
Luego, imponemos que al menos una sesi\'on sea larga:
\[
\sum_{t=1}^{T-K+1} z_t \;\geq\; 1.
\]
(Notar que $z_t$ solo tiene sentido para $t\leq T-K+1$; para los dem\'as $t$ podr\'iamos fijarlos a cero, o la suma con esos \'indices simplemente no los considera.)

\medskip
\noindent\textbf{(R7) Dominio de las variables.}
\[
x_t,\,s_t,\,e_t,\,z_t \;\in\; \{0,1\}, \qquad \forall\, t\in\mathcal{T}.
\]

\medskip

\noindent\emph{Observaci\'on sobre los bordes.} Las convenciones $x_0 := 0$ y $x_{T+1}:=0$ son claves. La primera hace que estudiar desde el bloque $1$ se considere un inicio. La segunda implementa literalmente lo que dice el enunciado: ``si se est\'a estudiando durante el \'ultimo bloque del d\'ia se considera que la sesi\'on acaba durante ese \'ultimo bloque''.

\section*{Pregunta 3}

\subsection{Soluci\'on gr\'afica}

Consideremos el problema:
\[
\max 2x_1 + x_2 \quad\text{s.a.}\quad x_1+x_2\leq 3{,}5,\; 2x_1+x_2\leq 6,\; x_1\leq 2{,}5,\; x_1,x_2\geq 0.
\]

\textbf{Regi\'on factible (con las tres restricciones).} Los v\'ertices son $(0,0)$, $(2{,}5,\,0)$, $(2{,}5,\,1)$ y $(0,\,3{,}5)$. Evaluando la funci\'on objetivo $z=2x_1+x_2$ en cada uno: $z(0,0)=0$, $z(2{,}5,0)=5$, $z(2{,}5,1)=6$, $z(0,\,3{,}5)=3{,}5$. El \'optimo es \emph{\'unico} y se alcanza en el v\'ertice $(x_1^*,x_2^*)=(2{,}5,\,1)$ con valor \'optimo $z^*=6$.

\begin{figure}[h!]
\centering
\includegraphics[width=0.72\textwidth]{grafico_p3a_original.pdf}
\caption{Regi\'on factible con las tres restricciones. El v\'ector $\nabla z=(2,1)$ indica la direcci\'on de crecimiento de la FO; la curva de nivel $z=6$ toca la regi\'on exactamente en $(2{,}5,\,1)$.}
\end{figure}

\textbf{Eliminando la restricci\'on $x_1\leq 2{,}5$.} Sin esta cota, los v\'ertices pasan a ser $(0,0)$, $(3,0)$, $(2{,}5,\,1)$ y $(0,\,3{,}5)$. Los valores de $z$ son ahora $0$, $6$, $6$ y $3{,}5$ respectivamente. Como el gradiente de la FO $(2,1)$ es paralelo al gradiente de la restricci\'on (2) $2x_1+x_2\leq 6$, la curva de nivel $z=6$ coincide con la arista entre $(2{,}5,\,1)$ y $(3,0)$. Por lo tanto aparecen \emph{\'optimos m\'ultiples}: \emph{cualquier} punto del segmento
\[
\{(x_1,x_2)\in\R^2_+ \,:\; 2x_1+x_2=6,\; 2{,}5\leq x_1\leq 3\}
\]
es \'optimo y el valor \'optimo sigue siendo $z^*=6$. Equivalentemente, el conjunto de \'optimos es
\[
\{(\lambda\cdot 2{,}5 + (1-\lambda)\cdot 3,\; \lambda\cdot 1 + (1-\lambda)\cdot 0) : \lambda\in[0,1]\} \;=\; \{(3-0{,}5\lambda,\,\lambda): \lambda\in[0,1]\}.
\]

\begin{figure}[h!]
\centering
\includegraphics[width=0.72\textwidth]{grafico_p3a_sin_r3.pdf}
\caption{Regi\'on factible sin la restricci\'on $x_1\leq 2{,}5$. La cara \'optima es el segmento que une $(2{,}5,\,1)$ con $(3,\,0)$.}
\end{figure}

\subsection{Implementaci\'on computacional}

\subsubsection*{1. Interpretaci\'on del modelo}

\textbf{Variables.} $y_i\in\{0,1\}$ indica si la bodega $i\in O$ es arrendada ($y_i=1$) o no ($y_i=0$). La variable continua $x_{ij}\geq 0$ representa las unidades de producto enviadas desde la bodega $i$ al cliente $j\in D$.

\smallskip
\textbf{Funci\'on objetivo.} Se minimiza el costo total, compuesto por (i) el costo variable de transporte $\sum_{i,j}c_{ij}x_{ij}$ y (ii) el costo fijo de arriendo de las bodegas utilizadas $\sum_i F_i y_i$.

\smallskip
\textbf{Restricciones.}
\begin{itemize}
    \item La restricci\'on (4) $\sum_j x_{ij}\leq s_i y_i$ cumple dos funciones simult\'aneamente: (a) impone la capacidad m\'axima $s_i$ cuando la bodega est\'a arrendada ($y_i=1$) y (b) prohibe enviar desde una bodega no arrendada ($y_i=0 \Rightarrow \sum_j x_{ij}\leq 0 \Rightarrow x_{ij}=0$ $\forall j$). Esta es la ``linealizaci\'on'' cl\'asica del acoplamiento abrir/usar una instalaci\'on.
    \item La restricci\'on (5) $\sum_i x_{ij}= d_j$ obliga a satisfacer \emph{exactamente} la demanda de cada cliente.
\end{itemize}
